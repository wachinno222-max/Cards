  website: "",                     /* e.g. https://binsense.com */
  social: [                        /* leave a link empty ("") to hide it */
    {label:"LinkedIn",  https://www.linkedin.com/in/opiyo-guido:""},
   
  ]
};
/* ================================================= */

const $ = id => document.getElementById(id);
document.title = C.name + " | Digital Card";
$("name").textContent = C.name;
$("title").textContent = C.title;
$("company").textContent = C.company;
$("loc").textContent = C.location ? "📍 " + C.location : "";
$("bio").textContent = C.bio;
$("foot").textContent = "Scan. Connect. Done.";
$("initials").textContent = C.name.split(" ").filter(Boolean).slice(0,2).map(w=>w[0]).join("").toUpperCase();
$("photo").src = C.photo;

const btns = [];
if (C.phone)    btns.push(`<a class="btn" href="tel:${C.phone}">📞 Call</a>`);
if (C.whatsapp) btns.push(`<a class="btn wa" href="https://wa.me/${C.whatsapp}" target="_blank" rel="noopener">💬 WhatsApp</a>`);
if (C.email)    btns.push(`<a class="btn" href="mailto:${C.email}">✉️ Email</a>`);
if (C.website)  btns.push(`<a class="btn" href="${C.website}" target="_blank" rel="noopener">🌐 Website</a>`);
btns.push(`<button class="btn save" onclick="saveContact()">⬇ Save to Contacts</button>`);
$("actions").innerHTML = btns.join("");

const links = C.social.filter(s => s.url);
$("social").innerHTML = links.map(s =>
  `<a class="chip" href="${s.url}" target="_blank" rel="noopener"><b>${s.label[0]}</b>${s.label}</a>`).join("");
if (!links.length) document.querySelector(".label").remove();

function saveContact(){
  const v = ["BEGIN:VCARD","VERSION:3.0","FN:"+C.name,"ORG:"+C.company,"TITLE:"+C.title,
    "TEL;TYPE=CELL:"+C.phone,"EMAIL:"+C.email,C.website?"URL:"+C.website:"",
    C.location?"ADR;TYPE=WORK:;;"+C.location+";;;;":"","END:VCARD"].filter(Boolean).join("\r\n");
  const a = document.createElement("a");
  a.href = URL.createObjectURL(new Blob([v],{type:"text/vcard"}));
  a.download = C.name.replace(/\s+/g,"_") + ".vcf";
  document.body.appendChild(a); a.click(); a.remove();
}
